client -> server:
    code:"<code>",data:"<data optional>

    codes:
        100 - key send:
            example:
                code:"100",data:"space"

        150 - file text write:
            example:
                code:"150",data:"hello"

        200 - press(and not release(hold)) a key:
            example:
                code:"200",data:"space"

        250 - release a key:
            example:
                code:"250",data:"space"

        300 - file keys:
            example:
                code:"300",data:"file"
        
        900 - logout:
            example:
                code:"900",data:""



server -> client:
    code:"<code>",data:"<data optional>"

    codes:
        100 - okay:
            example:
                code:"100",data:""
        
        300 - file keys:
            example:
                code:"300",data:"data.txt"

        500 - error:
            example:
                code:"500",data:"invalid key/message"
